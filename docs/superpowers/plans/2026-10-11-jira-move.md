# Jira Move Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Moving a Jira issue file to another project folder and committing moves the issue in Jira (with its sub-tasks), after showing what the move loses.

**Architecture:** The engine learns two generic things: resolving `Preview` actions with `Check` before the policy ask (class, detail, warning), and refreshing other resources an action changed (`Result.Refresh`). The Jira adapter switches to `DirTree`, plans a move from the target's createmeta and statuses (`move.go`), runs it through the bulk move API, polls the task, then applies the file's remaining changes against the moved issue.

**Tech Stack:** Go stdlib, existing `atlassian.Client`, `jtest` in-memory Jira, engine `fake` adapter.

**Spec:** `docs/superpowers/specs/2026-10-11-jira-move-design.md`

## Global Constraints

- Bulk move body: `sendBulkNotification: false`; `inferClassificationDefaults`, `inferFieldDefaults`, `inferSubtaskTypeDefault` true; `inferStatusDefaults` true only when no explicit `targetStatus`.
- Lossless move class `move`; lossy (type or status not kept, or any value dropped) class `delete`.
- Poll every 500 ms, 1 s after the first 5 s, up to 2 minutes.
- Messages, verbatim: `sub-tasks move with their parent: move <KEY> instead`; `<KEY> has no issue type "X" (types: …)`; `<KEY>'s <Type> workflow has no status "X" (statuses: …)`; `<KEY> requires <elem> (<Name>) for <Type> issues; add it to the file`; `move still running in Jira (task N); run gfs pull before committing again`; `will not appear in the customer portal: Request Type is empty`.
- `CustomerAdapter` stays `Flat`.
- Comments in code follow the repo: short, cite the spec section (`(jira move spec §5.2)`).

## Review Focus

1. A moved file that also carries a `<transitions>` block or a status edit the target workflow has: the move sets the status directly, no transition is sent afterwards (Task 5 test `TestMoveEditedStatus`).
2. A commit where the parent moves and its sub-task file was edited locally: the sub-task file is not refreshed over the edit (Task 2 test `TestRefreshSkipsChangedAndPending`).
3. A Check error on a preview action must not prompt and must not Apply (Task 1 test `TestPreviewCheckErrorFailsWithoutAsking`).
4. Retrying after a timed-out move that Jira finished later: no second bulk request, the other edits apply (Task 5 test `TestMoveAlreadyMoved`).
5. Bulk move 400 bodies `{"errors":[{"message":…}]}` must reach the user as the message, not raw JSON (Task 5 test `TestAPIErrorMessageList` in atlassian).

// checkRequest is what Check sees of fc.
func (e *Env) checkRequest(fc changes.FileChange) adapter.ApplyRequest {
	return adapter.ApplyRequest{
		Local: &adapter.Resource{ID: fc.ID, Path: fc.Path, Root: fc.Local.Content},
		Base:  baseResource(fc), Actions: fc.Actions, IDByPath: e.IDByPath, Open: e.open,
	}
}

// preview resolves fc's Preview actions with Check before the policy decides
// them (jira move spec §4.1). A Check error fails the file.
func (e *Env) preview(ctx context.Context, fc changes.FileChange) ([]adapter.Action, error) {
	checks := e.Session.Check(ctx, e.checkRequest(fc))
	acts := slices.Clone(fc.Actions)
	for i := range acts {
		if !acts[i].Preview || i >= len(checks) {
			continue
		}
		if checks[i].Err != nil {
			return nil, checks[i].Err
		}
		resolved(&acts[i], checks[i])
	}
	return acts, nil
}

// resolved takes what Check said about a: class, detail and warning.
func resolved(a *adapter.Action, c adapter.Result) {
	if c.Class != "" {
		a.Class = c.Class
	}
	if c.Detail != "" {
		a.Detail = c.Detail
	}
	if c.Warning != "" {
		a.Warning = c.Warning
	}
}

package engine

import (
	"context"
	"fmt"

	"github.com/KrzysztofBogdan/gitfs/internal/adapter"
	"github.com/KrzysztofBogdan/gitfs/internal/changes"
	"github.com/KrzysztofBogdan/gitfs/internal/envelope"
)

// refresh re-reads the resources fc's actions changed besides fc itself
// (jira move spec §4.2): the sub-tasks a move took along. Files with local
// changes, and files this commit has yet to handle, are left alone.
func (e *Env) refresh(ctx context.Context, fc changes.FileChange, results []adapter.Result, r *Report) error {
	for _, res := range results {
		if res.Err != nil {
			continue
		}
		for _, id := range res.Refresh {
			entry, ok := e.Index.ByID(id)
			if !ok || e.pending[id] {
				continue
			}
			if e.locallyChanged(entry.Path) {
				fmt.Fprintf(e.Out, "warning: %s: changed on the remote by this commit; run gfs pull\n", entry.Path)
				continue
			}
			remote, err := e.Session.Fetch(ctx, id)
			if err != nil {
				r.Failed++
				e.line("refresh", entry.Path, "", "FAIL", oneLine(err))
				e.Log("refresh", entry.Path, "", "FAIL", oneLine(err))
				continue
			}
			if err := e.Store(remote, entry.Path); err != nil {
				return err
			}
			if _, err := e.placeAttachments(id, remote.Path, remote.Root); err != nil {
				return err
			}
			e.line("refresh", entry.Path, remote.Path, "ok", "moved with "+fc.Path)
			e.Log("refresh", entry.Path, remote.Path, "ok", "moved with "+fc.Path)
		}
	}
	return nil
}

// locallyChanged reports whether the working file at p differs from its
// base, or is missing or unreadable.
func (e *Env) locallyChanged(p string) bool {
	s := e.Adapter.Schema()
	ldata, err1 := e.Tree.ReadFile(p)
	bdata, err2 := e.Tree.ReadBase(p)
	if err1 != nil || err2 != nil {
		return true
	}
	local, err1 := envelope.Parse(ldata)
	base, err2 := envelope.Parse(bdata)
	if err1 != nil || err2 != nil || local.Action != "" || len(local.Errors) > 0 || local.Conflict != nil {
		return true
	}
	return changes.CanonContent(local.Content, s) != changes.CanonContent(base.Content, s)
}

package jtest

import (
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"slices"
	"strings"
)

func init() { registrars = append(registrars, (*Server).moveRoutes) }

type moveTask struct {
	running   int
	fail      string
	processed []int
}

func (s *Server) moveRoutes(mux *http.ServeMux) {
	mux.HandleFunc("POST /rest/api/3/bulk/issues/move", s.bulkMove)
	mux.HandleFunc("GET /rest/api/3/bulk/queue/{id}", s.moveQueue)
}

type moveGroup struct {
	InferStatus  bool `json:"inferStatusDefaults"`
	TargetStatus []struct {
		Statuses map[string][]string `json:"statuses"`
	} `json:"targetStatus"`
	Issues []string `json:"issueIdsOrKeys"`
}

func (s *Server) bulkMove(w http.ResponseWriter, r *http.Request) {
	raw, _ := io.ReadAll(r.Body)
	var body struct {
		Mapping map[string]moveGroup `json:"targetToSourcesMapping"`
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	s.MoveBodies = append(s.MoveBodies, string(raw))
	if err := json.Unmarshal(raw, &body); err != nil {
		moveError(w, err.Error())
		return
	}
	type job struct {
		is     *Issue
		p      *Project
		typ    string
		status string
	}
	var jobs []job
	for key, g := range body.Mapping {
		pid, tid, _ := strings.Cut(key, ",")
		p := s.projectByID(pid)
		if p == nil || !slices.ContainsFunc(s.types[p.Key], func(t IssueType) bool { return t.ID == tid }) {
			moveError(w, "Invalid target group "+key)
			return
		}
		for _, k := range g.Issues {
			is := s.lookup(k)
			if is == nil {
				moveError(w, "Issue "+k+" not found")
				return
			}
			if s.isSubtask(is) {
				moveError(w, "Parent issue should also be moved or target Parent issue should be specified")
				return
			}
			status := ""
			if !g.InferStatus {
				wf := s.workflow(&Issue{Project: p.Key, Type: tid})
				for _, ts := range g.TargetStatus {
					for to := range ts.Statuses {
						i := slices.IndexFunc(wf.Statuses, func(st Status) bool { return st.ID == to })
						if i < 0 {
							moveError(w, fmt.Sprintf("You provided a target status that doesn't exist in the target workflow for the target group %s:%s.", pid, tid))
							return
						}
						status = wf.Statuses[i].Name
					}
				}
			}
			jobs = append(jobs, job{is, p, tid, status})
		}
	}
	id := s.nextID()
	t := &moveTask{running: s.MoveRunning, fail: s.MoveFail}
	if t.fail == "" {
		for _, j := range jobs {
			t.processed = append(t.processed, s.moveIssue(j.is, j.p, j.typ, j.status)...)
		}
	}
	s.tasks[id] = t
	w.WriteHeader(http.StatusCreated)
	writeJSON(w, map[string]any{"taskId": id})
}

// moveIssue moves is and its sub-tasks to p as type typ; status "" keeps
// the status by name, else the workflow's first. It returns the moved ids.
func (s *Server) moveIssue(is *Issue, p *Project, typ, status string) []int {
	old := is.Key
	s.relocate(is, p, typ, status)
	ids := []int{atoi(is.ID)}
	var subType string
	for _, t := range s.types[p.Key] {
		if t.Subtask {
			subType = t.ID
			break
		}
	}
	for _, c := range s.issues {
		if par, ok := c.Fields["parent"].(map[string]any); ok && par["key"] == old {
			c.Fields["parent"] = map[string]any{"key": is.Key}
			if s.isSubtask(c) {
				s.relocate(c, p, subType, "")
				ids = append(ids, atoi(c.ID))
			}
		}
	}
	return ids
}

func (s *Server) relocate(is *Issue, p *Project, typ, status string) {
	is.OldKeys = append(is.OldKeys, is.Key)
	s.counters[p.Key]++
	is.Project, is.Key, is.Type = p.Key, fmt.Sprintf("%s-%d", p.Key, s.counters[p.Key]), typ
	wf := s.workflow(is)
	switch {
	case status != "":
		is.Status = status
	case !slices.ContainsFunc(wf.Statuses, func(st Status) bool { return st.Name == is.Status }):
		is.Status = wf.Statuses[0].Name
	}
	fields := map[string]Field{}
	for _, f := range s.fields[p.Key+"/"+typ] {
		fields[f.ID] = f
	}
	for id, v := range is.Fields {
		f, ok := fields[id]
		switch {
		case id == "parent":
		case !ok:
			delete(is.Fields, id)
		case len(f.Options) > 0:
			if v = keepAllowed(v, f.Options); v == nil {
				delete(is.Fields, id)
			} else {
				is.Fields[id] = v
			}
		}
	}
	is.Updated = s.now()
}

// keepAllowed is v without the options or named values f does not allow.
func keepAllowed(v any, opts []Option) any {
	ok := func(m map[string]any) bool {
		return slices.ContainsFunc(opts, func(o Option) bool {
			return o.ID == m["id"] || o.Value == m["value"] || o.Value == m["name"]
		})
	}
	switch x := v.(type) {
	case map[string]any:
		if ok(x) {
			return x
		}
		return nil
	case []any:
		var kept []any
		for _, e := range x {
			if m, isMap := e.(map[string]any); isMap && ok(m) {
				kept = append(kept, e)
			}
		}
		if len(kept) == 0 {
			return nil
		}
		return kept
	}
	return v
}

	n := 0
	fmt.Sscan(s, &n)
	return n
}

// moveIndex is the index of the move action, -1 if none.
func moveIndex(acts []adapter.Action) int {
	for i, a := range acts {
		if a.Verb == "move" && a.Target == "" {
			return i
		}
	}
	return -1
}

func (m *movePlan) class() string {
	if !m.noop && len(m.notes)+len(m.losses) > 0 {
		return "delete"
	}
	return "move"
}

func (m *movePlan) detail() string {
	if m.noop {
		return "already in " + m.to.Key
	}
	status := m.statusName
	if m.statusTo == "" {
		status = m.to.Key + "'s default"
	}
	return fmt.Sprintf("move to %s as %s, status %s", m.to.Key, m.typ.Name, status)
}

func (m *movePlan) warning() string {
	if m.noop {
		return ""
	}
	parts := append([]string(nil), m.notes...)
	if len(m.losses) > 0 {
		parts = append(parts, "drops "+strings.Join(m.losses, ", "))
	}
	if m.to.JSM {
		parts = append(parts, "will not appear in the customer portal: Request Type is empty")
	}
	return strings.Join(parts, "; ")
}

// moveField is a target field as createmeta reports it, with its values.
type moveField struct {
	ID, Name             string
	Required, HasDefault bool
	Allowed              []allowedValue
}

type allowedValue struct{ ID, Name, Value string }

func (f moveField) allows(id, text string) bool {
	if len(f.Allowed) == 0 {
		return true // free values (labels, text) or a list createmeta does not give
	}
	for _, v := range f.Allowed {
		if (id != "" && v.ID == id) || strings.EqualFold(v.Name, text) || strings.EqualFold(v.Value, text) {
			return true
		}
	}
	return false
}

// moveFields reads the target type's create screen with allowed values;
// not cached: values change and a move is rare.
func (s *session) moveFields(ctx context.Context, key, typeID string) (map[string]moveField, error) {
	out := map[string]moveField{}
	for start := 0; ; {
		var resp struct {
			Fields []struct {
				FieldID    string         `json:"fieldId"`
				Name       string         `json:"name"`
				Required   bool           `json:"required"`
				HasDefault bool           `json:"hasDefaultValue"`
				Allowed    []allowedValue `json:"allowedValues"`
			} `json:"fields"`
			Total int `json:"total"`
		}
		path := fmt.Sprintf("/rest/api/3/issue/createmeta/%s/issuetypes/%s?startAt=%d&maxResults=200", url.PathEscape(key), url.PathEscape(typeID), start)
		if err := s.c.Do(ctx, http.MethodGet, path, nil, &resp); err != nil {
			return nil, fmt.Errorf("fields of %s type %s: %w", key, typeID, err)
		}

		}
		start += len(resp.Fields)
		if len(resp.Fields) == 0 || start >= resp.Total {
			return out, nil
		}
	}
}

// planMove works out how req's issue moves to the project of its new folder
// (jira move spec §5.2). It writes nothing.
func (s *session) planMove(ctx context.Context, req adapter.ApplyRequest) (*movePlan, error) {
	from, err := s.projectFor(req.Base.Path)
	if err != nil {
		return nil, err
	}
	to, err := s.projectFor(req.Local.Path)
	if err != nil {
		return nil, err
	}
	var cur moveState
	path := "/rest/api/3/issue/" + url.PathEscape(req.Base.ID) + "?fields=project,issuetype,status,parent"
	if err := s.c.Do(ctx, http.MethodGet, path, nil, &cur); err != nil {
		return nil, err
	}
	local, base := req.Local.Root, req.Base.Root
	m := &movePlan{id: req.Base.ID, key: cur.Key, from: from, to: to, statusFrom: cur.Fields.Status.ID, adjusted: local.Clone()}
	if strings.EqualFold(cur.Fields.Project.Key, to.Key) {
		m.noop = true
		return m, nil
	}
	if cur.Fields.Issuetype.Subtask {
		parent := ""
		if cur.Fields.Parent != nil {
			parent = cur.Fields.Parent.Key
		}
		return nil, fmt.Errorf("sub-tasks move with their parent: move %s instead", parent)
	}
	if to.Types == nil {
		if err := s.loadTypes(ctx, to); err != nil {
			return nil, err
		}
	}
	if err := m.chooseType(textOf(child(local, "type")), textOf(child(base, "type"))); err != nil {
		return nil, err
	}
	statuses, err := s.typeStatuses(ctx, to.Key, m.typ.ID)
	if err != nil {
		return nil, err
	}
	if err := m.chooseStatus(statuses, textOf(child(local, "status")), textOf(child(base, "status"))); err != nil {
		return nil, err
	}
	fields, err := s.moveFields(ctx, to.Key, m.typ.ID)
	if err != nil {
		return nil, err
	}
	m.drop(fields)
	if err := m.required(fields); err != nil {
		return nil, err
	}
	setText(m.adjusted, "type", m.typ.Name)
	if m.statusTo != "" {
		setText(m.adjusted, "status", m.statusName)
	}
	return m, nil
}

// chooseType picks the target type: the file's if the target has it, else
// (type not edited) Task or the standard type with the lowest id.
func (m *movePlan) chooseType(want, was string) error {
	if t := m.to.typeNamed(want); t != nil && !t.Subtask {
		m.typ = t
		return nil
	}
	var std []*typeMeta
	for _, t := range m.to.Types {
		if !t.Subtask {
			std = append(std, t)
		}
	}
	sort.Slice(std, func(i, j int) bool {
		a, b := std[i].ID, std[j].ID
		return len(a) < len(b) || (len(a) == len(b) && a < b)
	})
	if !strings.EqualFold(want, was) || len(std) == 0 {
		var names []string
		for _, t := range std {
			names = append(names, t.Name)
		}
		sort.Strings(names)
		return fmt.Errorf("%s has no issue type %q (types: %s)", m.to.Key, want, strings.Join(names, ", "))
	}
	m.typ = std[0]
	if t := m.to.typeNamed("Task"); t != nil && !t.Subtask {
		m.typ = t
	}
	m.notes = append(m.notes, fmt.Sprintf("type %s -> %s (%s has no %s)", want, m.typ.Name, m.to.Key, want))
	return nil
}

// chooseStatus maps to the file's status when the target workflow has it,
// else (status not edited) lets Jira pick the default.
func (m *movePlan) chooseStatus(sts []apiStatus, want, was string) error {
	var names []string
	for _, st := range sts {
		if strings.EqualFold(st.Name, want) {
			m.statusTo, m.statusName = st.ID, st.Name
			return nil
		}
		names = append(names, st.Name)
	}
	if !strings.EqualFold(want, was) {
		return fmt.Errorf("%s's %s workflow has no status %q (statuses: %s)", m.to.Key, m.typ.Name, want, strings.Join(names, ", "))
	}
	m.notes = append(m.notes, fmt.Sprintf("status %s -> %s's default", want, m.to.Key))
	return nil
}

// drop removes from the adjusted file what the target cannot hold and
// records each loss.
func (m *movePlan) drop(fields map[string]moveField) {
	var kept []*xmltree.Node
	for _, c := range m.adjusted.Children {
		if c.Kind != xmltree.Element || moveKept[c.Name] {
			kept = append(kept, c)
			continue
		}
		if c.Name == "field" {
			if m.dropField(c, fields) {
				kept = append(kept, c)
			}
			continue
		}
		id, ok := elemFields[c.Name]
		if !ok {
			kept = append(kept, c)
			continue
		}
		f, on := fields[id]
		switch {
		case c.Name == "components" || c.Name == "fixVersions" || c.Name == "affectsVersions":
			item := "version"
			if c.Name == "components" {
				item = "component"
			}
			var items []*xmltree.Node
			for _, it := range c.ChildrenNamed(item) {
				if on && f.allows("", textOf(it)) {
					items = append(items, it)
				} else {
					m.losses = append(m.losses, item+" "+textOf(it))
				}
			}
			if len(items) == 0 {
				continue
			}
			c.Children = itemsOnly(c, items)
		case !on:
			m.losses = append(m.losses, "<"+c.Name+">")
			continue
		case c.Name == "priority":
			if !f.allows("", textOf(c)) {
				m.losses = append(m.losses, "priority "+textOf(c))
				continue
			}
		}
		kept = append(kept, c)
	}
	m.adjusted.Children = kept
}

// itemsOnly keeps c's children that are not items, plus the kept items.
func itemsOnly(c *xmltree.Node, items []*xmltree.Node) []*xmltree.Node {
	var out []*xmltree.Node
	for _, ch := range c.Children {
		if ch.Kind != xmltree.Element {
			out = append(out, ch)
		}
	}
	return append(out, items...)
}

// dropField reports whether <field> c survives, dropping its options the
// target does not allow.
func (m *movePlan) dropField(c *xmltree.Node, fields map[string]moveField) bool {
	id, _ := c.Attr("id")
	name, _ := c.Attr("name")
	if name == "" {
		name = id
	}
	f, on := fields[id]
	if !on {
		m.losses = append(m.losses, "<field "+name+">")
		return false
	}
	opts := c.ChildrenNamed("option")
	if len(opts) == 0 {
		return true
	}
	var kept []*xmltree.Node
	for _, o := range opts {
		oid, _ := o.Attr("id")
		if f.allows(oid, textOf(o)) {
			kept = append(kept, o)
		} else {
			m.losses = append(m.losses, fmt.Sprintf("option %s of <field %s>", textOf(o), name))
		}
	}
	if len(kept) == 0 {
		return false
	}
	c.Children = itemsOnly(c, kept)
	return true
}

// required refuses a move that leaves a required target field without a
// value Jira could fill.
func (m *movePlan) required(fields map[string]moveField) error {
	var ids []string
	for id, f := range fields {
		switch id {
		case "project", "issuetype", "summary", "reporter", "parent":
			continue
		}
		if f.Required && !f.HasDefault && !hasValue(m.adjusted, id) {

}

// applyMove moves the issue, then applies the rest of the file against the
// moved issue (jira move spec §5.4).
func (s *session) applyMove(ctx context.Context, req adapter.ApplyRequest, mi int) []adapter.Result {
	out := results(req.Actions)
	m, err := s.planMove(ctx, req)
	if err != nil {
		return failAll(out, err)
	}
	var refresh []string
	if !m.noop {
		if refresh, err = s.bulkMove(ctx, m); err != nil {
			return failAll(out, err)
		}
	}
	is, err := s.getIssue(ctx, m.id)
	if err != nil {
		return failAll(out, err)
	}
	post, err := s.resource(ctx, is, true)
	if err != nil {
		return failAll(out, err)
	}
	adj := m.adjusted
	adj.SetAttr("key", is.Key)
	if !m.noop && m.statusTo == "" {
		setText(adj, "status", textOf(child(post.Root, "status")))
	}
	acts := changes.ResolveActions(&Adapter{}, post.Root, &envelope.Doc{Content: adj}, post.Path, post.Path)
	for _, a := range req.Actions {
		if a.IsAttachment() {
			acts = append(acts, a)
		}
	}
	inner := req
	inner.Local = &adapter.Resource{ID: m.id, Path: post.Path, Root: adj}
	inner.Base, inner.Actions = post, acts
	done := s.applyIssue(ctx, inner, results(acts))
	out[mi].Detail, out[mi].Refresh = is.Key, refresh
	matched := make([]bool, len(done))
	for i, a := range req.Actions {
		if i == mi {
			continue
		}
		for j, d := range done {
			if !matched[j] && sameAction(a, d.Action) {
				matched[j] = true
				out[i].Err, out[i].Code, out[i].ID, out[i].Version = d.Err, d.Code, d.ID, d.Version
				break
			}
		}
	}
	for j, d := range done { // re-set values the move dropped by id but the target has by name
		if !matched[j] && d.Err != nil {
			out = append(out, d)
		}
	}
	return out
}

func sameAction(a, b adapter.Action) bool {
	return a.Verb == b.Verb && a.Target == b.Target && a.Group == b.Group && a.File == b.File
}

// bulkMove submits the move and waits for it; it returns the ids of the
// other issues the move took along (its sub-tasks).
func (s *session) bulkMove(ctx context.Context, m *movePlan) ([]string, error) {
	group := map[string]any{
		"inferClassificationDefaults": true, "inferFieldDefaults": true, "inferSubtaskTypeDefault": true,
		"inferStatusDefaults": m.statusTo == "", "issueIdsOrKeys": []string{m.id},
	}
	if m.statusTo != "" {
		group["targetStatus"] = []any{map[string]any{"statuses": map[string][]string{m.statusTo: {m.statusFrom}}}}
	}
	body := map[string]any{"sendBulkNotification": false,
		"targetToSourcesMapping": map[string]any{m.to.ID + "," + m.typ.ID: group}}
	var resp struct {
		TaskID string `json:"taskId"`
	}
	if err := s.c.Do(ctx, http.MethodPost, "/rest/api/3/bulk/issues/move", body, &resp); err != nil {
		return nil, fmt.Errorf("move %s to %s: %w", m.key, m.to.Key, err)
	}
	return s.waitMove(ctx, m, resp.TaskID)
}

func (s *session) waitMove(ctx context.Context, m *movePlan, task string) ([]string, error) {
	var waited time.Duration
	for {
		var q struct {
			Status    string                     `json:"status"`
			Processed []json.Number              `json:"processedAccessibleIssues"`
			Failed    map[string]json.RawMessage `json:"failedAccessibleIssues"`
		}
		if err := s.c.Do(ctx, http.MethodGet, "/rest/api/3/bulk/queue/"+url.PathEscape(task), nil, &q); err != nil {
			return nil, fmt.Errorf("move %s to %s (task %s): %w", m.key, m.to.Key, task, err)
		}
		switch q.Status {
		case "COMPLETE":
			if msg, ok := q.Failed[m.id]; ok {
				return nil, fmt.Errorf("move %s to %s failed: %s", m.key, m.to.Key, failText(msg))
			}
			var others []string
			for _, n := range q.Processed {
				if n.String() != m.id {
					others = append(others, n.String())
				}
			}
			return others, nil
		case "FAILED", "CANCELLED", "DEAD":
			var msgs []string
			for _, raw := range q.Failed {
				msgs = append(msgs, failText(raw))
			}
			sort.Strings(msgs)
			return nil, fmt.Errorf("move %s to %s: task %s %s: %s", m.key, m.to.Key, task, strings.ToLower(q.Status), strings.Join(msgs, "; "))
		}
		if waited >= moveTimeout {
			return nil, fmt.Errorf("move still running in Jira (task %s); run gfs pull before committing again", task)
		}
		d := 500 * time.Millisecond
		if waited >= 5*time.Second {
			d = time.Second
		}
		if s.report != nil {
			s.report(adapter.Progress{Phase: "wait", Item: fmt.Sprintf("moving %s to %s", m.key, m.to.Key)})
		}
		if err := s.c.Sleep(ctx, d); err != nil {
			return nil, err
		}
		waited += d
	}
}

// failText is a failed issue's messages: a list of strings, or {"errors":[…]}.
func failText(raw json.RawMessage) string {
	var list []string
	if json.Unmarshal(raw, &list) == nil {
		return strings.Join(list, "; ")
	}
	var obj struct {
		Errors []string `json:"errors"`
	}
	if json.Unmarshal(raw, &obj) == nil && len(obj.Errors) > 0 {
		return strings.Join(obj.Errors, "; ")
	}
	return string(raw)
}
