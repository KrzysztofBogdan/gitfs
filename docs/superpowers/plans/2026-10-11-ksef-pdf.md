# KSeF PDF Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every invoice in a ksef mirror gets `<ksef>.pdf` beside its original, byte-identical to the KSeF web app's PDF, rendered by the Ministry's generator embedded in gfs and run in goja.

**Architecture:** A new package `internal/adapter/ksef/ksefpdf` embeds a gfs build of `ksef-pdf-generator` (made by `build.sh`), runs it in goja behind browser shims, derives the generator's inputs from the XML, and offers a `Renderer` and a `Pool`. The ksef adapter renders while listing and on `Download(…, "pdf")`, places `.pdf` files beside originals, and lists both under one `<file>` attachment element. The CLI gains `render` and `warn` progress phases.

**Tech Stack:** Go 1.25, `github.com/dop251/goja` at `39ec265`, `time/tzdata`; Node.js 22 + npm only inside `build.sh`.

**Spec:** `docs/superpowers/specs/2026-10-11-ksef-pdf-design.md` (builds on `docs/superpowers/specs/2026-10-09-ksef-invoices-design.md`).

## Global Constraints

* goja pinned to commit `39ec265` (2026-09-26); `go.mod` stays `go 1.25.0`.
* First embedded release: 1.1.40, commit `f59fc4e`.
* gfs never changes `time.Local`; local time inside the generator is Europe/Warsaw.
* Render time: `Naglowek/DataWytworzeniaFa` cut to whole ms; else midnight Europe/Warsaw of the KSeF number's date.
* KOD I: `<QRBase>/invoice/<Podmiot1 NIP>/<P_1 DD-MM-YYYY>/<SHA-256 Base64URL, no padding>`; QR bases `https://qr.ksef.mf.gov.pl`, `https://qr-test.ksef.mf.gov.pl`, `https://qr-demo.ksef.mf.gov.pl`.
* `acDate` = KSeF number date as `dd.mm.yyyy`.
* A render longer than 60 s is interrupted.
* Listing pool size `min(GOMAXPROCS, 4)`.
* Metadata attachment element is `file`; attributes `id name size sha256 generator`, all read-only.
* URL parameter `pdf=yes|no`, default yes, dropped from the canonical URL when yes.
* Copy: progress `Rendering %d/%d PDFs  <item>`; warning `warning: PDF of <ksef>: <error>`.
* Go code gofmt-clean before every commit (`gofmt -l` prints nothing).

## Review Focus

1. An invoice the generator cannot render (FA_RR, PEF, malformed XML) must not fail the clone: the invoice is mirrored without a PDF and one warning names it. Pinned in Task 6 (`TestListPDFFailureWarns`).
2. A machine whose time zone is not Poland must produce the same bytes. Pinned in Task 3 (`TestGolden` under `America/New_York`, with a delivery period that local getters format).
3. A UTF-16 or BOM-prefixed original must render. Pinned in Task 2 (`TestDecodeText`) and Task 3 (`TestRenderUTF16`).
4. Ctrl-C during a large clone must stop rendering promptly. Pinned in Task 4 (`TestPoolCancelled`) and Task 6 (`TestListCancelledWhileRendering`).
5. A user's own `scan.pdf` dropped in a month folder must stay a stray, not become an invoice's attachment. Pinned in Task 7 (the existing e2e stray check, kept).

```ts
import pdfMake from 'pdfmake/build/pdfmake';

export * from './index';

// gfsAddFont stores an already decoded font in pdfmake's virtual file
// system: with no encoding writeFileSync keeps the bytes as they are.
export function gfsAddFont(name: string, bytes: ArrayBuffer): void {
  (pdfMake as any).virtualfs.writeFileSync(name, new Uint8Array(bytes));
}
```

```js
const { execSync } = require('child_process');
const fs = require('fs');
const path = require('path');
let ls;
try {
  ls = execSync('npm ls --omit=dev --all --parseable', { encoding: 'utf8' });
} catch (e) {
  ls = e.stdout;
}
const seen = new Set();
const out = [];
for (const d of ls.trim().split('\n').slice(1)) {
  const pj = JSON.parse(fs.readFileSync(path.join(d, 'package.json'), 'utf8'));

out.push('== Roboto fonts (Apache-2.0)\nCopyright Google LLC. Licensed under the Apache License, Version 2.0:\nhttps://www.apache.org/licenses/LICENSE-2.0');
```

```bash
# Rebuilds assets/ and the golden PDFs in testdata/ from a commit of the
# Ministry's ksef-pdf-generator (ksef pdf spec §3.2). Needs git, Node.js 22
# and npm; nothing else in gfs does.

set -euo pipefail
commit=${1:?usage: build.sh <commit of github.com/CIRFMF/ksef-pdf-generator>}
here=$(cd "$(dirname "$0")" && pwd)
work=$(mktemp -d)
trap 'rm -rf "$work"' EXIT
git clone -q https://github.com/CIRFMF/ksef-pdf-generator.git "$work/gen"
cd "$work/gen"
git checkout -q "$commit"
npm ci --no-audit --no-fund >/dev/null

# Golden PDFs from the unchanged source, as the web app renders them.
cp "$here/build/golden.spec.ts" src/gfs-golden.spec.ts
TESTDATA="$here/testdata" TZ=Europe/Warsaw npx vitest run src/gfs-golden.spec.ts >/dev/null
rm src/gfs-golden.spec.ts

# The gfs build: fonts come from Go.
cp "$here/build/gfs-entry.ts" "$here/build/gfs-empty-vfs.js" src/
node "$here/build/patch-vite.cjs" vite.config.ts

mkdir -p "$here/assets"
cp dist/ksef-fe-invoice-converter.umd.cjs "$here/assets/generator.js"

cp LICENSE "$here/assets/LICENSE"
```

```bash
python3 -I - <<'EOF'
import json, re, hashlib, base64, datetime
def case(xml, pdf, ksef):
    b = open('internal/adapter/ksef/ksefpdf/testdata/' + xml, 'rb').read(); s = b.decode('utf-8-sig')
    nip = re.search(r'<(?:\w+:)?Podmiot1\b.*?<(?:\w+:)?NIP>\s*(\d+)', s, re.S).group(1)
    y, m, d = re.search(r'<(?:\w+:)?P_1>\s*(\d{4})-(\d\d)-(\d\d)', s).groups()
    created = re.search(r'DataWytworzeniaFa>([^<]+)<', s).group(1)
    whole, _, frac = created.rstrip('Z').partition('.')
    t = datetime.datetime.strptime(whole, '%Y-%m-%dT%H:%M:%S').replace(tzinfo=datetime.timezone.utc)
    ms = int(t.timestamp()) * 1000 + int((frac + '000')[:3])
    h = base64.urlsafe_b64encode(hashlib.sha256(b).digest()).decode().rstrip('=')
    kd = ksef.split('-')[1]
    return dict(xml=xml, pdf=pdf, ksef=ksef, qrBase='https://qr.ksef.mf.gov.pl', clockMs=ms,
                acDate=f'{kd[6:]}.{kd[4:6]}.{kd[:4]}', qrCode=f'https://qr.ksef.mf.gov.pl/invoice/{nip}/{d}-{m}-{y}/{h}')
json.dump([case('fa3.xml', 'fa3.pdf', '1234563218-20260810-0100A1B2C3D4-5E'),
           case('upstream-example.xml', 'upstream-example.pdf', '5555555555-20250808-9231003CA67B-BE')],
```

```go
func utf16Bytes(s string, little, bom bool) []byte {
	if bom {
		s = "\ufeff" + s
	}
	var out []byte
	for _, u := range utf16.Encode([]rune(s)) {
		if little {
			out = append(out, byte(u), byte(u>>8))
		} else {
			out = append(out, byte(u>>8), byte(u))
		}
	}
	return out
}

func TestDecodeText(t *testing.T) {

		"utf-16le bom":   utf16Bytes(want, true, true),
		"utf-16le":       utf16Bytes(want, true, false),
		"utf-16be bom":   utf16Bytes(want, false, true),
		"utf-16be":       utf16Bytes(want, false, false),
	} {
		if got := decodeText(b); got != want {
			t.Errorf("%s: %q", name, got)
		}
	}
	if got := decodeText([]byte("<a>\xff</a>")); got != "<a>\ufffd</a>" {
		t.Errorf("invalid UTF-8 becomes U+FFFD: %q", got)

}
```

```go
package ksefpdf

import (
	"crypto/sha256"
	"encoding/base64"
	"encoding/binary"
	"encoding/xml"
	"io"
	"strings"
	"time"
	_ "time/tzdata" // Europe/Warsaw on systems without a zone database
	"unicode/utf16"
)

// Input is one invoice to render (ksef pdf spec §4.1).
type Input struct {
	XML        []byte // the original, as KSeF stores it
	KsefNumber string
	QRBase     string // https://qr.ksef.mf.gov.pl, -test or -demo; "" draws no QR code
}

// facts are what the generator gets besides the KSeF number.
type facts struct {
	text    string // the XML as the browser's readAsText gives it
	acDate  string // dd.mm.yyyy from the KSeF number (§4.1)
	qrCode  string // KOD I link (§4.4), "" when it cannot be built
	clockMs int64  // render time (§4.3)
}

var warsaw = func() *time.Location {
	l, err := time.LoadLocation("Europe/Warsaw")
	if err != nil {
		panic(err)
	}
	return l
}()

func factsOf(in Input) facts {
	f := facts{text: decodeText(in.XML)}
	nip, p1, created := fields(f.text)
	day, hasDay := ksefDay(in.KsefNumber)
	if hasDay {
		f.acDate = day.Format("02.01.2006")
		f.clockMs = day.UnixMilli()
	}
	if t, ok := parseCreated(created); ok {
		f.clockMs = t.UnixMilli()
	}
	if d, err := time.Parse("2006-01-02", p1); err == nil && nip != "" && in.QRBase != "" {
		h := sha256.Sum256(in.XML)
		f.qrCode = in.QRBase + "/invoice/" + nip + "/" + d.Format("02-01-2006") + "/" + base64.RawURLEncoding.EncodeToString(h[:])
	}
	return f
}

// ksefDay is midnight Europe/Warsaw of the date in a KSeF number
// (<nip>-<YYYYMMDD>-<technical>-<check>).
func ksefDay(k string) (time.Time, bool) {
	parts := strings.Split(k, "-")
	if len(parts) < 2 {
		return time.Time{}, false
	}
	t, err := time.ParseInLocation("20060102", parts[1], warsaw)
	return t, err == nil
}

// parseCreated reads DataWytworzeniaFa: with a zone, or as Europe/Warsaw
// local time without one.
func parseCreated(s string) (time.Time, bool) {
	if t, err := time.Parse(time.RFC3339Nano, s); err == nil {
		return t, true
	}
	if t, err := time.ParseInLocation("2006-01-02T15:04:05.999999999", s, warsaw); err == nil {
		return t, true
	}
	return time.Time{}, false
}

// fields reads Podmiot1/DaneIdentyfikacyjne/NIP, Fa/P_1 and
// Naglowek/DataWytworzeniaFa under the root, by local name.
func fields(text string) (nip, p1, created string) {
	d := xml.NewDecoder(strings.NewReader(text))
	d.CharsetReader = func(_ string, r io.Reader) (io.Reader, error) { return r, nil } // already text
	var stack []string
	for {
		tok, err := d.Token()
		if err != nil {
			return
		}
		switch t := tok.(type) {
		case xml.StartElement:
			stack = append(stack, t.Name.Local)
		case xml.EndElement:
			stack = stack[:len(stack)-1]
		case xml.CharData:
			if len(stack) < 2 {
				continue
			}
			v := strings.TrimSpace(string(t))
			switch strings.Join(stack[1:], "/") {
			case "Podmiot1/DaneIdentyfikacyjne/NIP":
				nip = first(nip, v)
			case "Fa/P_1":
				p1 = first(p1, v)
			case "Naglowek/DataWytworzeniaFa":
				created = first(created, v)
			}
		}
	}
}

// first keeps the first value found.
func first(old, v string) string {
	if old != "" {
		return old
	}
	return v
}

// decodeText reads the XML the way FileReader.readAsText does after the
// generator's encoding check: UTF-16 by byte-order mark or a zero byte
// beside the first '<', otherwise UTF-8; a byte-order mark is dropped.
func decodeText(b []byte) string {
	switch {
	case len(b) >= 2 && (b[0] == 0xff && b[1] == 0xfe || b[0] == '<' && b[1] == 0):
		return utf16Text(b, binary.LittleEndian)
	case len(b) >= 2 && (b[0] == 0xfe && b[1] == 0xff || b[0] == 0 && b[1] == '<'):
		return utf16Text(b, binary.BigEndian)
	}
	return strings.TrimPrefix(strings.ToValidUTF8(string(b), "\ufffd"), "\ufeff")
}

func utf16Text(b []byte, order binary.ByteOrder) string {
	u := make([]uint16, len(b)/2)
	for i := range u {
		u[i] = order.Uint16(b[2*i:])
	}
	return strings.TrimPrefix(string(utf16.Decode(u)), "\ufeff")
}
```

```go
// Renderer is one goja runtime with the generator loaded and its fonts
// added (ksef pdf spec §4.1). It is not safe for concurrent use.
type Renderer struct {
	vm             *goja.Runtime
	gen, runTimers goja.Callable

// NewRenderer starts a runtime: about 0.5 s and 140 MB.
func NewRenderer() (*Renderer, error) {
	prog, err := compiled()
	if err != nil {
		return nil, fmt.Errorf("ksef-pdf-generator: %w", err)
	}
	vm := goja.New()
	vm.Set("__now", int64(0))
	vm.Set("__waw", func(ms int64) []any {
		t := time.UnixMilli(ms).In(warsaw)
		_, off := t.Zone()
		return []any{t.Year(), int(t.Month()) - 1, t.Day(), t.Hour(), t.Minute(), t.Second(), t.Nanosecond() / 1e6, int(t.Weekday()), -off / 60}
	})
	vm.Set("__wawLocal", func(y, m0, d, h, mi, s, ms int64) int64 {
		return time.Date(int(y), time.Month(m0+1), int(d), int(h), int(mi), int(s), int(ms)*1e6, warsaw).UnixMilli()
	})
	vm.Set("__dtf", func(ms int64, withTime, withSeconds bool) string {
		layout := "02.01.2006"
		if withTime {
			layout += ", 15:04"
			if withSeconds {
				layout += ":05"
			}
		}
		return time.UnixMilli(ms).In(warsaw).Format(layout)
	})
	vm.Set("__encode", func(s string) goja.ArrayBuffer { return vm.NewArrayBuffer([]byte(s)) })
	vm.Set("__decode", func(b goja.Value, enc string) (string, error) { return decodeJS(b, enc) })
	if _, err := vm.RunScript("shim.js", shimJS); err != nil {
		return nil, fmt.Errorf("ksef-pdf-generator shims: %w", err)
	}
	if _, err := vm.RunProgram(prog); err != nil {
		return nil, fmt.Errorf("ksef-pdf-generator: %w", err)
	}

		return nil, errors.New("ksef-pdf-generator: the bundle defines no library")

// Render renders one invoice at its render time (§4.3).
func (r *Renderer) Render(in Input) ([]byte, error) {
	return r.render(context.Background(), in, clockFromXML)
}

// render renders at clockMs, or at §4.3's time for clockFromXML. ctx and
// renderTimeout interrupt it; an interrupted renderer is broken for good.
func (r *Renderer) render(ctx context.Context, in Input, clockMs int64) ([]byte, error) {
	if r.broken {
		return nil, errBroken
	}
	f := factsOf(in)
	if clockMs == clockFromXML {
		clockMs = f.clockMs
	}
	r.vm.Set("__now", clockMs)
	timer := time.AfterFunc(renderTimeout, func() { r.vm.Interrupt(errTimeout) })
	stop := context.AfterFunc(ctx, func() { r.vm.Interrupt(ctx.Err()) })
	pdf, err := r.run(f, in.KsefNumber)
	timer.Stop()
	stop()
	r.vm.ClearInterrupt()
	var ie *goja.InterruptedError
	if errors.As(err, &ie) {
		r.broken = true
		if ie.Value() == errTimeout {
			return nil, fmt.Errorf("the generator ran longer than %v", renderTimeout)
		}
		if e, ok := ie.Value().(error); ok {
			return nil, e
		}
		return nil, err
	}
	return pdf, err
}

func (r *Renderer) run(f facts, ksef string) ([]byte, error) {
	vm := r.vm
	file, err := vm.New(vm.Get("File"), vm.ToValue(f.text))
	if err != nil {
		return nil, err
	}
	add := vm.NewObject()
	add.Set("nrKSeF", ksef)
	if f.acDate != "" {
		add.Set("acDate", f.acDate)
	}
	if f.qrCode != "" {
		add.Set("qrCode", f.qrCode)
	}
	res, err := r.gen(goja.Undefined(), file, add, vm.ToValue("base64"))
	if err != nil {
		return nil, jsError(err)
	}
	p, ok := res.Export().(*goja.Promise)
	if !ok {
		return nil, errors.New("generateInvoice returned no promise")
	}
	for p.State() == goja.PromiseStatePending {
		n, err := r.runTimers(goja.Undefined())
		if err != nil {
			return nil, jsError(err)
		}
		if n.ToInteger() == 0 && p.State() == goja.PromiseStatePending {
			return nil, errors.New("the generator stopped without finishing")
		}
	}
	if p.State() == goja.PromiseStateRejected {
		return nil, errors.New(jsMessage(p.Result()))
	}
	return base64.StdEncoding.DecodeString(p.Result().String())
}

// jsError turns a JavaScript exception into its message; interrupts pass.
func jsError(err error) error {
	var ex *goja.Exception
	if errors.As(err, &ex) {
		return errors.New(jsMessage(ex.Value()))
	}
	return err
}

func jsMessage(v goja.Value) string {
	if o, ok := v.(*goja.Object); ok {
		if m := o.Get("message"); m != nil && !goja.IsUndefined(m) {
			return m.String()
		}
	}
	return v.String()
}

// decodeJS is TextDecoder.decode for the encodings fontkit and the
// generator use.
func decodeJS(b goja.Value, enc string) (string, error) {
	var bs []byte
	switch v := b.Export().(type) {
	case goja.ArrayBuffer:
		bs = v.Bytes()
	case []byte:
		bs = v
	default:
		return "", fmt.Errorf("TextDecoder: cannot decode %T", v)
	}
	switch strings.ToLower(enc) {
	case "utf-8", "utf8", "":
		return strings.ToValidUTF8(string(bs), "\ufffd"), nil
	case "utf-16be", "utf-16le", "utf-16":
		u := make([]uint16, len(bs)/2)
		for i := range u {
			if strings.ToLower(enc) == "utf-16be" {
				u[i] = uint16(bs[2*i])<<8 | uint16(bs[2*i+1])
			} else {
				u[i] = uint16(bs[2*i]) | uint16(bs[2*i+1])<<8
			}
		}
		return string(utf16.Decode(u)), nil
	case "ascii", "latin1", "iso-8859-1", "us-ascii":
		r := make([]rune, len(bs))
		for i, c := range bs {
			r[i] = rune(c)
		}
		return string(r), nil
	}
	return "", fmt.Errorf("TextDecoder: unsupported encoding %q", enc)
}
```

```go
package ksefpdf

import (
	"context"
	"sync"
	"sync/atomic"
)

// Pool shares at most n renderers between goroutines (ksef pdf spec §4.5).
type Pool struct {
	sem     chan struct{} // a token per render in progress
	mu      sync.Mutex
	idle    []*Renderer
	created atomic.Int32
}

func NewPool(n int) *Pool { return &Pool{sem: make(chan struct{}, max(n, 1))} }

// Render renders on an idle renderer, or a new one while fewer than n
// exist. ctx cancels the wait and interrupts the render.
func (p *Pool) Render(ctx context.Context, in Input) ([]byte, error) {
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	select {
	case p.sem <- struct{}{}:
	case <-ctx.Done():
		return nil, ctx.Err()
	}
	defer func() { <-p.sem }()
	r, err := p.get()
	if err != nil {
		return nil, err
	}
	pdf, err := r.render(ctx, in, clockFromXML)
	p.put(r)
	return pdf, err
}

func (p *Pool) get() (*Renderer, error) {
	p.mu.Lock()
	if k := len(p.idle); k > 0 {
		r := p.idle[k-1]
		p.idle = p.idle[:k-1]
		p.mu.Unlock()
		return r, nil
	}
	p.mu.Unlock()
	p.created.Add(1)
	return NewRenderer()
}

func (p *Pool) put(r *Renderer) {
	if r.broken {
		return
	}
	p.mu.Lock()
	p.idle = append(p.idle, r)
	p.mu.Unlock()
}

// Close drops the idle renderers; their memory goes with the next GC.
func (p *Pool) Close() {
	p.mu.Lock()
	p.idle = nil
	p.mu.Unlock()
}
```

```go
package ksef

import (
	"context"
	"fmt"
	"sync"

	"github.com/KrzysztofBogdan/gitfs/internal/adapter"
	"github.com/KrzysztofBogdan/gitfs/internal/adapter/ksef/ksefpdf"
)

// roleInvoice is one invoice of a listing in one role.
type roleInvoice struct {
	role string
	inv  invoice
}

// renderAll renders one PDF per distinct invoice (ksef pdf spec §5.2). A
// failed render is reported as a warning and leaves that invoice without a
// PDF; only a cancelled context fails.
func (s *session) renderAll(ctx context.Context, found []roleInvoice) (map[string][]byte, error) {
	out := map[string][]byte{}
	if !s.t.pdf {
		return out, nil
	}
	var todo []invoice
	seen := map[string]bool{}
	for _, f := range found {
		if k := f.inv.meta.KsefNumber; !seen[k] {
			seen[k] = true
			todo = append(todo, f.inv)
		}
	}
	if len(todo) == 0 {
		return out, nil
	}
	pool := ksefpdf.NewPool(s.pdfWorkers)
	defer pool.Close()
	var mu sync.Mutex
	var wg sync.WaitGroup
	done := 0
	s.report(adapter.Progress{Phase: "render", Total: len(todo)})
	for _, inv := range todo {
		wg.Add(1)
		go func() {
			defer wg.Done()
			k := inv.meta.KsefNumber
			pdf, err := pool.Render(ctx, ksefpdf.Input{XML: inv.xml, KsefNumber: k, QRBase: qrBase[s.t.host]})
			mu.Lock()
			defer mu.Unlock()
			if ctx.Err() != nil {
				return
			}
			done++
			if err != nil {
				s.report(adapter.Progress{Phase: "warn", Item: fmt.Sprintf("PDF of %s: %v", k, err)})
			} else {
				out[k] = pdf
			}
			s.report(adapter.Progress{Phase: "render", Done: done, Total: len(todo), Item: k})
		}()
	}
	wg.Wait()
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	return out, nil
}
```

```go
// Download reads an original again (gfs get), checked against its hash;
// for the PDF it renders that original (ksef pdf spec §5.3).
func (s *session) Download(ctx context.Context, resID, attID string, w io.Writer) (adapter.AttachmentInfo, error) {
	_, k, ok := strings.Cut(resID, "/")
	if !ok || (attID != "fa" && (attID != "pdf" || !s.t.pdf)) {
		return adapter.AttachmentInfo{}, adapter.ErrNotFound
	}
	resp, data, err := s.c.send(ctx, http.MethodGet, "/invoices/ksef/"+url.PathEscape(k), true, nil, true)
	if err != nil {
		return adapter.AttachmentInfo{}, err
	}
	if h := resp.Header.Get("X-Ms-Meta-Hash"); h != "" && h != b64sha(data) {
		return adapter.AttachmentInfo{}, fmt.Errorf("invoice %s: content does not match KSeF's hash", k)
	}
	if attID == "pdf" {
		pool := ksefpdf.NewPool(1)
		defer pool.Close()
		if data, err = pool.Render(ctx, ksefpdf.Input{XML: data, KsefNumber: k, QRBase: qrBase[s.t.host]}); err != nil {
			return adapter.AttachmentInfo{}, fmt.Errorf("PDF of %s: %w", k, err)
		}
	}
	if _, err := w.Write(data); err != nil {
		return adapter.AttachmentInfo{}, err
	}
	return adapter.AttachmentInfo{Version: "-", Size: int64(len(data))}, nil
}
```
